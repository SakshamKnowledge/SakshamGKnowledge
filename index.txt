    <script>
        let currentClass = 1;

        // Sample Questions Database
        const sampleData = {
            1: [
                { q: "Water boiled directly from lakes is always safe to drink.", type: "True/False", a: "False (Water must be purified or boiled first)." },
                { q: "Which organ helps us see things around us?", type: "MCQ", a: "Eyes" },
                { q: "Name a basic need of all living plants.", type: "Short Answer", a: "Sunlight and Water" }
            ],
            2: [
                { q: "Plants produce oxygen during photosynthesis.", type: "True/False", a: "True" },
                { q: "Which part of the plant absorbs water from the soil?", type: "MCQ", a: "Roots" }
            ]
        };

        function loadClassData(classNum) {
            currentClass = classNum;
            document.getElementById('classTitle').innerText = `Class ${classNum} Study Material`;
            
            const container = document.getElementById('questionsContainer');
            container.innerHTML = '';

            const questions = sampleData[classNum] || [
                { q: `Sample Science Question for Class ${classNum}?`, type: "General Science", a: "Standard concept explanation for this grade." },
                { q: `Sample Science GK Question for Class ${classNum}?`, type: "Science GK", a: "General Knowledge answer related to nature and space." }
            ];

            questions.forEach((item, idx) => {
                const card = document.createElement('div');
                card.className = "p-4 border rounded-xl bg-slate-50 hover:bg-blue-50/30 transition";
                card.innerHTML = `
                    <div class="flex items-center justify-between mb-2">
                        <span class="text-xs font-semibold px-2.5 py-1 bg-blue-100 text-blue-800 rounded-full">Q${idx+1} • ${item.type}</span>
                    </div>
                    <p class="font-medium text-slate-800 text-base mb-2">${item.q}</p>
                    <p class="text-sm text-slate-600 bg-white p-3 rounded-lg border border-slate-200"><span class="font-semibold text-emerald-600">Answer:</span> ${item.a}</p>
                `;
                container.appendChild(card);
            });
        }

        // PDF Generation with hardcoded 'saksham' watermark
        function downloadPDF() {
            const { jsPDF } = window.jspdf;
            const doc = new jsPDF();

            // Header Title
            doc.setFontSize(18);
            doc.setTextColor(30, 58, 138);
            doc.text(`Saksham Knowledge - Class ${currentClass} Questions`, 14, 20);

            // Questions Content
            doc.setFontSize(11);
            doc.setTextColor(50, 50, 50);
            
            const questions = sampleData[currentClass] || [
                { q: `Sample Science Question for Class ${currentClass}?`, a: "Standard concept explanation for this grade." },
                { q: `Sample Science GK Question for Class ${currentClass}?`, a: "General Knowledge answer related to nature and space." }
            ];

            let y = 35;
            questions.forEach((q, idx) => {
                doc.setFont("helvetica", "bold");
                doc.text(`Q${idx+1}: ${q.q}`, 14, y);
                y += 7;
                doc.setFont("helvetica", "normal");
                doc.text(`Ans: ${q.a}`, 18, y);
                y += 12;
            });

            // Watermark (Hardcoded 'saksham' only)
            doc.setFontSize(10);
            doc.setTextColor(150, 150, 150);
            doc.text("saksham", 14, 285);

            doc.save(`Saksham_Knowledge_Class_${currentClass}.pdf`);
        }

        // Modal Controls & Login
        function openModal() { document.getElementById('loginModal').classList.remove('hidden'); }
        function closeModal() { document.getElementById('loginModal').classList.add('hidden'); }

        function handleLogin(e) {
            e.preventDefault();
            const u = document.getElementById('usernameInput').value.trim();
            const p = document.getElementById('passwordInput').value.trim();

            if (u === "343642760015" && p === "Manju@1234") {
                alert("Founder Login Successful!");
                closeModal();
            } else {
                document.getElementById('loginError').classList.remove('hidden');
            }
        }

        // Initial Load
        loadClassData(1);
    </script>
